ยังไม่มีบัญชีผู้ใช้ในระบบ ให้สร้างบัญชีใน Supabase ก่อนเข้าใช้:

  1. เปิด Supabase Authentication → Users แล้วเพิ่มผู้ใช้ด้วยอีเมลและรหัสผ่านที่ต้องการ
  2. คัดลอก UUID ของผู้ใช้ แล้วแทน UUID และอีเมลใน SQL ด้านล่าง
insert into public.users (id, email, display_name, role, active) values ('', '', 'ผู้ดูแลระบบ', 'admin', true);
